        function alternarPermissaoModulo(indexUsuario, chaveCurso) {
            const usuario = usuariosPermitidos[indexUsuario];
            if (!usuario.modulosPermitidos) usuario.modulosPermitidos = [];
            const pos = usuario.modulosPermitidos.indexOf(chaveCurso);
            if (pos > -1) usuario.modulosPermitidos.splice(pos, 1);
            else usuario.modulosPermitidos.push(chaveCurso);
            salvarUsuarios();
        }

        function iniciarQuiz() {
            const containerQuiz = document.getElementById('quizContainer');

        function mostrarQuestao() {
            const container = document.getElementById('quizContainer');
            if (!container) return;
            container.innerHTML = '';

            if (indiceAtual < questoesAtuais.length) {
                const item = questoesAtuais[indiceAtual];
                const div = document.createElement('div');
                div.className = 'bg-slate-50 dark:bg-darkBlue-900 border-l-4 border-amber-500 dark:border-brandYellow rounded-xl p-4 mb-4 shadow-sm';

                const progresso = document.createElement('div');
                progresso.className = 'text-center font-bold text-slate-500 dark:text-slate-400 mb-3 text-xs uppercase';
                progresso.innerText = `Pergunta ${indiceAtual + 1} de 5`;
                container.appendChild(progresso);

                const pPergunta = document.createElement('p');
                pPergunta.className = 'font-medium text-slate-800 dark:text-slate-200 mb-3';
                pPergunta.innerText = item.p;
                div.appendChild(pPergunta);

                const respostasEmbaralhadas = [...item.r].sort(() => 0.5 - Math.random());
                respostasEmbaralhadas.forEach(res => {
                    const label = document.createElement('label');
                    label.className = 'block my-2 cursor-pointer p-2.5 rounded-lg border border-slate-200 dark:border-darkBlue-700 bg-white dark:bg-darkBlue-800 hover:bg-amber-50 dark:hover:bg-darkBlue-700/50 text-sm text-slate-700 dark:text-slate-300 transition';

                    const input = document.createElement('input');
                    input.type = 'radio';
                    input.name = 'perguntaAtual';
                    input.value = res;
                    input.className = 'mr-2 accent-amber-500';

                    label.appendChild(input);
                    label.append(res);
                    div.appendChild(label);
                });

                container.appendChild(div);

                const btn = document.createElement('button');
                btn.className = 'w-full mt-2 py-3 bg-brandYellow hover:bg-brandYellow-light text-darkBlue-900 font-bold rounded-lg text-sm transition shadow';
                btn.innerText = indiceAtual === questoesAtuais.length - 1 ? 'Finalizar e Ver Resultado' : 'Próxima Pergunta';
                btn.onclick = avancarProxima;
                container.appendChild(btn);
            } else {
                exibirResultadoFinal();
            }
        }

        function avancarProxima() {
            const selecionada = document.querySelector('input[name="perguntaAtual"]:checked');

            mostrarQuestao();

        function exibirResultadoFinal() {
